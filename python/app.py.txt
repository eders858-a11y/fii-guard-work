@app.route('/api/proventos-lote', methods=['GET'])
def get_proventos_lote():
    tickers_param = request.args.get('tickers', '')
    tickers = [t.strip().upper() for t in tickers_param.split(',') if t.strip()]
    
    all_dividends = []
    
    for ticker_name in tickers:
        dividends_history = None
        
        # Gera variações automáticas com e sem .SA para garantir a compatibilidade
        variations = []
        clean_name = ticker_name.replace('.SA', '')
        variations.append(f"{clean_name}.SA")
        variations.append(clean_name)
        
        # Tenta buscar em cada variação até encontrar dados válidos
        for symbol in variations:
            try:
                ticker_obj = yf.Ticker(symbol)
                hist = ticker_obj.dividends
                if hist is not None and not hist.empty:
                    dividends_history = hist
                    break
            except Exception:
                continue
        
        if dividends_history is not None and not dividends_history.empty:
            dividends_df = dividends_history.reset_index()
            date_col = 'Date' if 'Date' in dividends_df.columns else dividends_df.columns[0]
            val_col = 'Dividends' if 'Dividends' in dividends_df.columns else dividends_df.columns[1]
            
            dividends_df[date_col] = pd.to_datetime(dividends_df[date_col])
            dividends_df = dividends_df.sort_values(by=date_col, ascending=True)
            
            for _, row in dividends_df.iterrows():
                d_val = float(row[val_col])
                if d_val > 0:  # Ignora valores zerados
                    all_dividends.append({
                        'ticker': clean_name,  # Mantém o ticker limpo padronizado para o app
                        'dataCom': row[date_col].strftime('%Y-%m-%d'),
                        'dataPagamento': row[date_col].strftime('%Y-%m-%d'),
                        'tipo': 'Rendimento',
                        'valorUnitario': round(d_val, 4)
                    })
                    
    return jsonify({"dividends": all_dividends}), 200

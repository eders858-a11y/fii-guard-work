from flask import Flask, jsonify, request
import yfinance as yf

app = Flask(__name__)

@app.route('/api/proventos-lote', methods=['GET'])
def get_proventos_lote():
    tickers_param = request.args.get('tickers', '').upper().strip()
    if not tickers_param:
        return jsonify({'error': 'Nenhum ticker informado'}), 400

    lista_tickers = [t.strip() for t in tickers_param.split(',') if t.strip()]
    resultado_geral = {}

    for ticker in lista_tickers:
        lista_divs = []
        try:
            simbolo = ticker if '.' in ticker else f"{ticker}.SA"
            ativo = yf.Ticker(simbolo)
            
            acoes = ativo.get_actions()
            
            if acoes is not None and not acoes.empty:
                if "Dividends" in acoes.columns:
                    divs = acoes[acoes["Dividends"] > 0]
                    for data, row in divs.iterrows():
                        data_str = str(data).split('T')[0].split(' ')[0]
                        lista_divs.append({
                            'ticker': ticker.replace('.SA', ''),
                            'dataCom': data_str,
                            'dataPagamento': data_str,
                            'valorUnitario': float(row["Dividends"]),
                            'tipo': 'Rendimento'
                        })
        except Exception as e:
            print(f"Erro ao buscar dividendos reais para {ticker}: {e}")
        
        resultado_geral[ticker.replace('.SA', '')] = lista_divs

    return jsonify({'dividends': resultado_geral})

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
